# --- Phase 5: API surface ----------------------------------------------------
# Upgraded from 0.121.2 for security: that release pins starlette <0.50, and
# every starlette in that range carries PYSEC-2026-161 (a Host header can make
# `request.url.path` disagree with the routed path, bypassing any middleware
# that reads it), plus PYSEC-2026-248/249/2280/2281. 0.141.1 drops the upper
# bound, which is what allows the patched starlette below.
fastapi==0.141.1
# Pinned explicitly rather than left to fastapi's resolver: the whole point of
# the fastapi upgrade is to reach a starlette at or above 1.3.1, and an
# unpinned transitive dependency would let a fresh install quietly resolve
# back to a vulnerable one.
starlette==1.6.0
uvicorn[standard]==0.41.0
# httpx backs fastapi.testclient; no test makes a network call.
httpx==0.28.1
# Loads .env for local development. Optional at runtime -- config falls back to
# the process environment when it is absent (app/backend/core/config.py).
python-dotenv==1.2.3

# --- Phase 5: real LLM provider ----------------------------------------------
# Needed only for LLM_PROVIDER=real. The deterministic provider -- and the

openai==2.14.0
